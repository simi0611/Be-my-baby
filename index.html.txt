  <button id="yes">YES 💖</button>
  <button id="no">NO 🙄</button>

  <script>
    const noBtn = document.getElementById("no");
    const message = document.getElementById("message");

    const sass = [
      "Excuse me??? 😒",
      "Wrong button bestie 💅",
      "Why you running? 🏃‍♂️",
      "Just say yes already 😤",
      "You KNOW the answer 😏",
      "Stop playing with my heart 💔"
    ];

    let count = 0;

    // NO button runs away on tap or hover (tablet friendly)
    function moveNo() {
      const x = Math.random() * (window.innerWidth - 200);
      const y = Math.random() * (window.innerHeight - 200);
      noBtn.style.left = `${x}px`;
      noBtn.style.top = `${y}px`;
      message.innerText = sass[count % sass.length];
      count++;
    }

    noBtn.addEventListener("mouseover", moveNo);
    noBtn.addEventListener("click", moveNo);

    document.getElementById("yes").addEventListener("click", () => {
      document.body.innerHTML = `
        <h1 style="font-size:4em;color:white;">YAYYY 💖</h1>
        <p style="font-size:2.2em;color:white;">
          I knew you’d say yes 😘<br>
          Happy Valentine’s Day, my love 💘
        </p>
      `;
    });

    function setFont(font) {
      document.body.className = font;
    }
  </script>
